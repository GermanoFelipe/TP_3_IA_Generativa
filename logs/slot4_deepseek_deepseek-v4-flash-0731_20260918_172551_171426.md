#!/usr/bin/env python3
import sys

def main():
    if len(sys.argv) != 3:
        sys.stderr.write("Uso: python3 vida.py <archivo_estado_inicial> <generaciones>\n")
        sys.exit(1)
    
    archivo = sys.argv[1]
    try:
        generaciones = int(sys.argv[2])
    except ValueError:
        sys.stderr.write("El parametro generaciones debe ser un entero no negativo\n")
        sys.exit(1)
    
    if generaciones < 0:
        sys.stderr.write("El parametro generaciones debe ser no negativo\n")
        sys.exit(1)
    
    try:
        with open(archivo, 'r') as f:
            lineas = f.readlines()
    except Exception as e:
        sys.stderr.write(f"Error al leer archivo: {e}\n")
        sys.exit(1)
    
    # Limpiar solo saltos de línea (retorno de carro y nueva línea)
    grilla = []
    for linea in lineas:
        fila = linea.replace('\r', '').replace('\n', '')
        grilla.append(list(fila))
    
    if len(grilla) == 0:
        return
    
    alto = len(grilla)
    ancho = len(grilla[0])
    
    def contar_vecinas(filas, r, c):
        count = 0
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                if dr == 0 and dc == 0:
                    continue
                nr = r + dr
                nc = c + dc
                if 0 <= nr < alto and 0 <= nc < ancho:
                    if filas[nr][nc] == '#':
                        count += 1
        return count
    
    for _ in range(generaciones):
        nueva_grilla = []
        for r in range(alto):
            nueva_fila = []
            for c in range(ancho):
                celda = grilla[r][c]
                vecinas = contar_vecinas(grilla, r, c)
                if celda == '#':
                    nueva = '#' if vecinas in (2, 3) else '.'
                else:
                    nueva = '#' if vecinas == 3 else '.'
                nueva_fila.append(nueva)
            nueva_grilla.append(nueva_fila)
        grilla = nueva_grilla
    
    # Imprimir cada fila con un nuevo salto de línea al final
    for fila in grilla:
        print(''.join(fila))

if __name__ == '__main__':
    main()
