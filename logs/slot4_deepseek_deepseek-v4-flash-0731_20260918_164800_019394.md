import sys


def read_grid(path):
    with open(path, "r", encoding="utf-8") as f:
        return [line.rstrip("\r\n") for line in f]


def next_generation(grid, height, width):
    new_grid = []

    for r in range(height):
        new_row = []

        for c in range(width):
            neighbors = 0

            for dr in (-1, 0, 1):
                nr = r + dr
                if nr < 0 or nr >= height:
                    continue

                for dc in (-1, 0, 1):
                    if dr == 0 and dc == 0:
                        continue

                    nc = c + dc
                    if nc < 0 or nc >= width:
                        continue

                    if grid[nr][nc] == "#":
                        neighbors += 1

            if grid[r][c] == "#":
                if neighbors == 2 or neighbors == 3:
                    new_row.append("#")
                else:
                    new_row.append(".")
            else:
                if neighbors == 3:
                    new_row.append("#")
                else:
                    new_row.append(".")

        new_grid.append("".join(new_row))

    return new_grid


def main():
    if len(sys.argv) != 3:
        sys.stderr.write("usage: python3 vida.py <archivo_estado_inicial> <generaciones>\n")
        return 2

    try:
        generations = int(sys.argv[2])
    except ValueError:
        sys.stderr.write("error: generaciones debe ser un entero\n")
        return 2

    if generations < 0:
        sys.stderr.write("error: generaciones debe ser no negativo\n")
        return 2

    try:
        grid = read_grid(sys.argv[1])
    except OSError as e:
        sys.stderr.write("error: no se pudo leer archivo: " + str(e) + "\n")
        return 1

    height = len(grid)
    if height == 0:
        return 0

    width = len(grid[0])

    for _ in range(generations):
        grid = next_generation(grid, height, width)

    sys.stdout.write("\n".join(grid) + "\n")
    return 0


if __name__ == "__main__":
    sys.exit(main())
